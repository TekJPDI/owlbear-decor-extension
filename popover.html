OBR.onReady(() => { OBR.broadcast.onMessage("DECOR_EVENT", (event) => { if (event.data.action === "SHOW") { OBR.modal.open({ url: `https://tekjpdi.github.io/owlbear-decor-extension/display.html?img=${encodeURIComponent(event.data.url)}`, fullScreen: true }); } else if (event.data.action === "CLOSE") { OBR.modal.close(); } }); document.getElementById('addBtn').addEventListener('click', () => { const title = document.getElementById('decorTitle').value; const url = document.getElementById('decorUrl').value; if (title && url) { decors.push({ title, url }); renderList(); document.getElementById('decorTitle').value = ''; document.getElementById('decorUrl').value = ''; } }); document.getElementById('closeAllBtn').addEventListener('click', () => { OBR.broadcast.sendMessage("DECOR_EVENT", { action: "CLOSE" }); OBR.modal.close(); }); }); function renderList() { const container = document.getElementById('decorList'); container.innerHTML = ''; decors.forEach((item, index) => { const div = document.createElement('div'); div.className = 'decor-item'; div.innerHTML = ` ${item.title} `; container.appendChild(div); }); } function showDecor(url) { OBR.broadcast.sendMessage("DECOR_EVENT", { action: "SHOW", url: url }); OBR.modal.open({ url: `https://tekjpdi.github.io/owlbear-decor-extension/display.html?img=${encodeURIComponent(url)}`, fullScreen: true }); }
